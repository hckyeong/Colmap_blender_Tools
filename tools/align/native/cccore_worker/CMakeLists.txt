cmake_minimum_required(VERSION 3.24)
project(colmap_cccore_worker LANGUAGES CXX)
set(CCCORELIB_USE_QT_CONCURRENT OFF CACHE BOOL "" FORCE)
set(CCCORELIB_USE_TBB ON CACHE BOOL "" FORCE)
set(CCCORELIB_USE_CGAL OFF CACHE BOOL "" FORCE)
set(CCCORELIB_SHARED ON CACHE BOOL "" FORCE)
set(COLMAP_TBB_VERSION "2022.0.0")
find_package(TBB ${COLMAP_TBB_VERSION} EXACT COMPONENTS tbb CONFIG REQUIRED)
if(NOT "${TBB_VERSION}" VERSION_EQUAL "${COLMAP_TBB_VERSION}")
  message(FATAL_ERROR "oneTBB ${COLMAP_TBB_VERSION} 정확한 버전이 필요합니다. 현재 버전: ${TBB_VERSION}")
endif()
set(COLMAP_TBB_VERSION "${TBB_VERSION}")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/bin")
if(NOT DEFINED CCCORE_SOURCE OR NOT DEFINED JSON_INCLUDE_DIR OR NOT DEFINED CORE_COMMIT)
  message(FATAL_ERROR "CCCORE_SOURCE, JSON_INCLUDE_DIR, CORE_COMMIT 경로와 식별자가 필요합니다.")
endif()
add_subdirectory("${CCCORE_SOURCE}" cccore)
if(NOT TARGET TBB::tbb)
  message(FATAL_ERROR "oneTBB의 TBB::tbb 대상을 찾지 못했습니다.")
endif()
get_target_property(CCCORE_COMPILE_DEFINITIONS CCCoreLib COMPILE_DEFINITIONS)
list(FIND CCCORE_COMPILE_DEFINITIONS "CC_CORE_LIB_USES_TBB" CCCORE_TBB_DEFINITION_INDEX)
if(CCCORE_TBB_DEFINITION_INDEX EQUAL -1)
  message(FATAL_ERROR "CCCoreLib의 TBB 컴파일 정의가 활성화되지 않았습니다.")
endif()
get_target_property(CCCORE_LINK_LIBRARIES CCCoreLib LINK_LIBRARIES)
list(FIND CCCORE_LINK_LIBRARIES "TBB::tbb" CCCORE_TBB_LINK_INDEX)
if(CCCORE_TBB_LINK_INDEX EQUAL -1)
  message(FATAL_ERROR "CCCoreLib의 oneTBB 링크가 활성화되지 않았습니다.")
endif()
add_executable(colmap_cccore_worker main.cpp)
target_compile_features(colmap_cccore_worker PRIVATE cxx_std_17)
target_include_directories(colmap_cccore_worker PRIVATE "${JSON_INCLUDE_DIR}")
target_link_libraries(colmap_cccore_worker PRIVATE CCCoreLib::CCCoreLib Psapi)
target_compile_definitions(colmap_cccore_worker PRIVATE NOMINMAX CORE_COMMIT="${CORE_COMMIT}" CORE_PATCH="${CORE_PATCH}" TBB_VERSION_PIN="${COLMAP_TBB_VERSION}")
if(MSVC)
  target_compile_options(colmap_cccore_worker PRIVATE /utf-8 /W4 /permissive-)
  target_compile_options(CCCoreLib PRIVATE /utf-8)
endif()
install(TARGETS colmap_cccore_worker RUNTIME DESTINATION .)
