cmake_minimum_required(VERSION 3.24)
project(colmap_cccore_worker LANGUAGES CXX)
set(CCCORELIB_USE_QT_CONCURRENT OFF CACHE BOOL "" FORCE)
set(CCCORELIB_USE_TBB ON CACHE BOOL "" FORCE)
set(CCCORELIB_USE_CGAL OFF CACHE BOOL "" FORCE)
set(CCCORELIB_SHARED ON CACHE BOOL "" FORCE)
set(COLMAP_TBB_VERSION "2022.0.0")
find_package(TBB ${COLMAP_TBB_VERSION} EXACT COMPONENTS tbb CONFIG REQUIRED)
if(NOT "${TBB_VERSION}" VERSION_EQUAL "${COLMAP_TBB_VERSION}")
  message(FATAL_ERROR "oneTBB ${COLMAP_TBB_VERSION} 정확한 버전이 필요합니다. 현재 버전: ${TBB_VERSION}")
endif()
set(COLMAP_TBB_VERSION "${TBB_VERSION}")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/bin")
if(NOT DEFINED CCCORE_SOURCE OR NOT DEFINED JSON_INCLUDE_DIR OR NOT DEFINED CORE_COMMIT)
  message(FATAL_ERROR "CCCORE_SOURCE, JSON_INCLUDE_DIR, CORE_COMMIT 경로와 식별자가 필요합니다.")
endif()
add_subdirectory("${CCCORE_SOURCE}" cccore)
if(NOT TARGET TBB::tbb)
  message(FATAL_ERROR "oneTBB의 TBB::tbb 대상을 찾지 못했습니다.")
endif()
get_target_property(CCCORE_COMPILE_DEFINITIONS CCCoreLib COMPILE_DEFINITIONS)
list(FIND CCCORE_COMPILE_DEFINITIONS "CC_CORE_LIB_USES_TBB" CCCORE_TBB_DEFINITION_INDEX)
if(CCCORE_TBB_DEFINITION_INDEX EQUAL -1)
  message(FATAL_ERROR "CCCoreLib의 TBB 컴파일 정의가 활성화되지 않았습니다.")
endif()
get_target_property(CCCORE_LINK_LIBRARIES CCCoreLib LINK_LIBRARIES)
list(FIND CCCORE_LINK_LIBRARIES "TBB::tbb" CCCORE_TBB_LINK_INDEX)
if(CCCORE_TBB_LINK_INDEX EQUAL -1)
  message(FATAL_ERROR "CCCoreLib의 oneTBB 링크가 활성화되지 않았습니다.")
endif()
add_executable(colmap_cccore_worker main.cpp mesh_bvh.cpp)
target_compile_features(colmap_cccore_worker PRIVATE cxx_std_17)
target_include_directories(colmap_cccore_worker PRIVATE "${JSON_INCLUDE_DIR}")
target_link_libraries(colmap_cccore_worker PRIVATE CCCoreLib::CCCoreLib TBB::tbb Psapi)
target_compile_definitions(colmap_cccore_worker PRIVATE NOMINMAX CORE_COMMIT="${CORE_COMMIT}" CORE_PATCH="${CORE_PATCH}" TBB_VERSION_PIN="${COLMAP_TBB_VERSION}")
if(MSVC)
  target_compile_options(colmap_cccore_worker PRIVATE /utf-8 /W4 /permissive-)
  target_compile_options(CCCoreLib PRIVATE /utf-8)
endif()
install(TARGETS colmap_cccore_worker RUNTIME DESTINATION .)

# 메시 검사는 기존 정렬 엔진과 독립된 호출 규약과 라이브러리로 제공합니다.
add_library(colmap_mesh_cleanup SHARED mesh_cleanup.cpp)
target_compile_features(colmap_mesh_cleanup PRIVATE cxx_std_17)
target_link_libraries(colmap_mesh_cleanup PRIVATE TBB::tbb)
target_compile_definitions(colmap_mesh_cleanup PRIVATE NOMINMAX)
set_target_properties(colmap_mesh_cleanup PROPERTIES PREFIX "" OUTPUT_NAME "colmap_mesh_cleanup")
if(MSVC)
  target_compile_options(colmap_mesh_cleanup PRIVATE /utf-8 /W4 /permissive- /fp:strict)
else()
  target_compile_options(colmap_mesh_cleanup PRIVATE -fno-fast-math -ffp-contract=off)
endif()
install(TARGETS colmap_mesh_cleanup RUNTIME DESTINATION . LIBRARY DESTINATION .)

option(COLMAP_BUILD_BVH_TESTS "BVH 정확 표면 검색 단위 시험 빌드" OFF)
if(COLMAP_BUILD_BVH_TESTS)
  # 배포 원문의 시험 경로를 먼저 찾고 저장소 배치 또는 명시한 경로를 사용합니다.
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../tests/align/native/test_mesh_bvh.cpp")
    set(COLMAP_BVH_DEFAULT_TEST_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../tests/align/native")
  else()
    set(COLMAP_BVH_DEFAULT_TEST_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../tests/align/native")
  endif()
  set(COLMAP_BVH_TEST_SOURCE_DIR "${COLMAP_BVH_DEFAULT_TEST_DIR}" CACHE PATH "BVH 독립 시험 원문 경로")
  add_executable(test_mesh_bvh "${COLMAP_BVH_TEST_SOURCE_DIR}/test_mesh_bvh.cpp" mesh_bvh.cpp)
  target_compile_features(test_mesh_bvh PRIVATE cxx_std_17)
  target_include_directories(test_mesh_bvh PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  target_link_libraries(test_mesh_bvh PRIVATE CCCoreLib::CCCoreLib TBB::tbb)
  target_compile_definitions(test_mesh_bvh PRIVATE NOMINMAX)

  target_compile_features(test_bvh_icp PRIVATE cxx_std_17)
  target_include_directories(test_bvh_icp PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${JSON_INCLUDE_DIR}")
  target_link_libraries(test_bvh_icp PRIVATE CCCoreLib::CCCoreLib TBB::tbb)
  target_compile_definitions(test_bvh_icp PRIVATE NOMINMAX)
  if(MSVC)
    target_compile_options(test_mesh_bvh PRIVATE /utf-8 /W4 /permissive-)
