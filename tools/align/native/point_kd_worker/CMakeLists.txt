cmake_minimum_required(VERSION 3.24)
project(colmap_point_kd_worker LANGUAGES CXX)

find_package(TBB 2022.0.0 EXACT COMPONENTS tbb CONFIG REQUIRED)
if(NOT DEFINED EIGEN_INCLUDE_DIR OR NOT EXISTS "${EIGEN_INCLUDE_DIR}/Eigen/Core")
  message(FATAL_ERROR "고정된 Eigen 3.4.0의 EIGEN_INCLUDE_DIR가 필요합니다.")
endif()
if(NOT DEFINED JSON_INCLUDE_DIR OR NOT EXISTS "${JSON_INCLUDE_DIR}/json.hpp")
  message(FATAL_ERROR "고정된 nlohmann JSON 3.11.3의 JSON_INCLUDE_DIR가 필요합니다.")
endif()
file(SHA256 "${JSON_INCLUDE_DIR}/json.hpp" JSON_HEADER_SHA256)
if(NOT JSON_HEADER_SHA256 STREQUAL "9bea4c8066ef4a1c206b2be5a36302f8926f7fdc6087af5d20b417d0cf103ea6")
  message(FATAL_ERROR "JSON 헤더가 잠금 해시와 다릅니다.")
endif()
file(READ "${EIGEN_INCLUDE_DIR}/Eigen/src/Core/util/Macros.h" EIGEN_VERSION_SOURCE)
if(NOT EIGEN_VERSION_SOURCE MATCHES "#define EIGEN_WORLD_VERSION 3" OR
   NOT EIGEN_VERSION_SOURCE MATCHES "#define EIGEN_MAJOR_VERSION 4" OR
   NOT EIGEN_VERSION_SOURCE MATCHES "#define EIGEN_MINOR_VERSION 0")
  message(FATAL_ERROR "Eigen 3.4.0만 지원합니다.")
endif()

add_executable(colmap_point_kd_worker main.cpp point_kd.cpp)
target_compile_features(colmap_point_kd_worker PRIVATE cxx_std_17)
target_include_directories(colmap_point_kd_worker SYSTEM PRIVATE "${EIGEN_INCLUDE_DIR}" "${JSON_INCLUDE_DIR}")
target_link_libraries(colmap_point_kd_worker PRIVATE TBB::tbb)
target_compile_definitions(colmap_point_kd_worker PRIVATE NOMINMAX EIGEN_DONT_PARALLELIZE)
set_target_properties(colmap_point_kd_worker PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_BINARY_DIR}/bin")
if(MSVC)
  target_compile_options(colmap_point_kd_worker PRIVATE /utf-8 /W4 /permissive- /fp:strict)
else()
  target_compile_options(colmap_point_kd_worker PRIVATE -Wall -Wextra -fno-fast-math -ffp-contract=off)
endif()
add_custom_command(TARGET colmap_point_kd_worker POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:TBB::tbb>" "$<TARGET_FILE_DIR:colmap_point_kd_worker>")
install(TARGETS colmap_point_kd_worker RUNTIME DESTINATION .)
install(FILES "$<TARGET_FILE:TBB::tbb>" DESTINATION .)
